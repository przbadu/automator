   ```python
   @observe(name="explorer_analyze_document")
   async def _execute_analyze_document(
       document_id: str, question: str, user_id: str, db, client, model, provider
   ) -> str:
       """Run the document analysis sub-agent and collect its output."""
       # Look up filename
       cursor = await db.execute(
           "SELECT filename FROM documents WHERE id = ? AND user_id = ?",
           (document_id, user_id)
       )
       row = await cursor.fetchone()
       if not row:
           return f"Document with ID '{document_id}' not found."

       # Collect sub-agent output (non-streaming for explorer consumption)
       result_parts = []
       async for event in run_sub_agent(
           user_message=question,
           user_id=user_id,
           chat_history=[],
           document_id=document_id,
           document_filename=row[0],
           client=client,
           model=model,
           provider=provider,
           db=db,
       ):
           if event["type"] == "delta":
               result_parts.append(event.get("content", ""))

       result = "".join(result_parts)
       # Truncate to prevent context blowup
       if len(result) > 4000:
           result = result[:4000] + "\n\n[Output truncated at 4000 characters]"
       return result
   ```
